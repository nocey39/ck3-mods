dau_upgrade_status_text = {
	type = character
	text = {
		trigger = {
			NOT = { has_variable = dau_up_enabled }
		}
		localization_key = dau_upgrade_status_off
	}
	text = {
		trigger = {
			var:dau_up_group_1 = 0
			var:dau_up_group_2 = 0
			var:dau_up_group_3 = 0
		}
		localization_key = dau_upgrade_status_no_scope
	}
	text = {
		trigger = {
			var:dau_up_last_count > 0
		}
		localization_key = dau_upgrade_status_done
	}
	text = {
		trigger = {
			var:dau_up_status = 2
		}
		localization_key = dau_upgrade_status_none
	}
	text = {
		localization_key = dau_upgrade_status_ready
	}
}
