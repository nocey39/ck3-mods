ama_acc_culture_trigger = {
    trigger_if = { limit = { $TYPE$ = flag:vanguard_attribute } culture ?= { OR = { has_innovation = innovation_quilted_armor has_cultural_tradition = tradition_fp1_coastal_warriors has_cultural_tradition = tradition_hird has_cultural_tradition = tradition_futuwaa has_cultural_tradition = tradition_druzhina has_cultural_tradition = tradition_khadga_puja has_cultural_tradition = tradition_garuda_warriors has_cultural_tradition = tradition_himalayan_settlers has_cultural_tradition = tradition_mubarizuns has_cultural_tradition = tradition_burman_royal_army has_cultural_tradition = tradition_mountaineer_ruralism has_innovation = innovation_sarawit has_innovation = innovation_legionnaires has_cultural_parameter = unlock_zhanmadao has_cultural_parameter = unlock_burenjia } } }
    trigger_if = { limit = { $TYPE$ = flag:lancer_attribute } culture ?= { OR = { has_innovation = innovation_arched_saddle has_cultural_tradition = tradition_caucasian_wolves has_cultural_tradition = tradition_roman_legacy has_innovation = innovation_valets has_cultural_tradition = tradition_ep3_audacious_cadets has_cultural_tradition = tradition_ep3_imperial_tagmata has_cultural_parameter = unlock_maa_cataphract_archers has_innovation = innovation_tiefutu has_cultural_parameter = unlock_maa_black_armor_cavalry } } }
    trigger_if = { limit = { $TYPE$ = flag:crossbowmen_attribute } culture ?= { OR = { has_innovation = innovation_advanced_bowmaking has_innovation = innovation_repeating_crossbow } } }
    trigger_if = { limit = { $TYPE$ = flag:besieger_attribute } culture ?= { has_innovation = innovation_catapult } }
    trigger_if = { limit = { $TYPE$ = flag:camelry_attribute } culture ?= { has_innovation = innovation_war_camels } }
    trigger_if = { limit = { $TYPE$ = flag:elephantry_attribute } culture ?= { has_innovation = innovation_elephantry } }
    trigger_if = { limit = { $TYPE$ = flag:horse_archer_attribute } culture ?= { OR = { has_cultural_parameter = unlock_maa_horse_archers has_cultural_parameter = unlock_maa_mangudai has_cultural_parameter = unlock_emishi_horse_archers_units has_cultural_parameter = unlock_mounted_samurai_units has_cultural_parameter = unlock_maa_cataphract_archers } } }
    trigger_if = { limit = { $TYPE$ = flag:gunpowder_attribute } culture = { OR = { has_innovation = innovation_gunpowder has_innovation = innovation_fire_medicine } } }
    trigger_if = { limit = { $TYPE$ = flag:huntsmaster_attribute } culture = { NOT = { has_cultural_tradition = tradition_vegetarianism } } }
    trigger_if = { limit = { $TYPE$ = flag:house_knight_attribute } dynasty = { has_dynasty_perk = kin_legacy_4 } }
}
