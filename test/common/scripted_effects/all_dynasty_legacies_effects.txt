# Preserve the Renown balance when granting a legacy.
all_dynasty_legacy_perk_effect = {
	if = {
		limit = { NOT = { has_dynasty_perk = $PERK$ } }
		save_scope_value_as = { name = adl_renown_before_grant value = dynasty_prestige }
		add_dynasty_perk = $PERK$
		if = {
			limit = { dynasty_prestige < scope:adl_renown_before_grant }
			add_dynasty_prestige = {
				value = scope:adl_renown_before_grant
				subtract = dynasty_prestige
			}
		}
	}
}

adl_open_next_choice_effect = {
	dynasty = { save_scope_as = adl_choice_dynasty }
	if = {
		limit = { adl_can_choose_ce1_heroic_legacy_3_trigger = yes }
		add_character_flag = { flag = adl_choice_open days = 7 }
		trigger_event = { id = adl_choices.0001 }
	}
	else_if = {
		limit = { adl_can_choose_ce1_heroic_legacy_4_trigger = yes }
		add_character_flag = { flag = adl_choice_open days = 7 }
		trigger_event = { id = adl_choices.0002 }
	}
	else_if = {
		limit = { adl_can_choose_ce1_legitimacy_legacy_5_trigger = yes }
		add_character_flag = { flag = adl_choice_open days = 7 }
		trigger_event = { id = adl_choices.0003 }
	}
	else_if = {
		limit = { adl_can_choose_blood_legacy_4_trigger = yes }
		add_character_flag = { flag = adl_choice_open days = 7 }
		trigger_event = { id = adl_choices.0004 }
	}
}
