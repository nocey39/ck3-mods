court_auto_toggle_memory_gui = {
    scope = character
    is_shown = { has_variable = court_auto_memory_enabled }
    is_valid = { is_ai = no is_alive = yes }
    effect = {
        court_auto_initialize_effect = yes
        court_auto_memory_reset_observations_effect = yes
        if = {
            limit = { has_variable = court_auto_memory_enabled }
            remove_variable = court_auto_memory_enabled
        }
        else = {
            set_variable = court_auto_memory_enabled
            court_auto_tasks_clear_exhausted_effect = yes
        }
        court_auto_restart_schedule_effect = yes
    }
}
