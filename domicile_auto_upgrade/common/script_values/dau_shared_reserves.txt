dau_shared_gold_reserve = {
    value = 0
    if = {
        limit = {
            has_domicile = yes
            domicile = {
                has_variable = dau_enabled
                has_variable = dau_gold_reserve
            }
        }
        add = domicile.var:dau_gold_reserve
    }
    if = {
        limit = {
            has_variable = dau_land_enabled
            has_variable = dau_land_gold_reserve
        }
        min = var:dau_land_gold_reserve
    }
}
