dejure_auto_direct_target_tier_trigger = { OR = {
        AND = { tier = tier_county root = { var:dejure_auto_direct_tier = 1 } }
        AND = { tier = tier_duchy root = { var:dejure_auto_direct_tier = 2 } }
        AND = { tier = tier_kingdom root = { var:dejure_auto_direct_tier = 3 } }
        AND = { tier = tier_empire root = { var:dejure_auto_direct_tier = 4 } }
    } }

dejure_auto_direct_last_state_matches_trigger = {
    exists = var:dejure_auto_manual_last_title
    OR = {
        AND = { var:dejure_auto_manual_mode = 0
            OR = {
                AND = { var:dejure_auto_direct_tier = 1
                    is_target_in_variable_list = { name = dejure_auto_manual_counties target = var:dejure_auto_manual_last_title } }
                AND = { var:dejure_auto_direct_tier >= 2 var:dejure_auto_direct_tier <= 4
                    is_target_in_variable_list = { name = dejure_auto_manual_titles target = var:dejure_auto_manual_last_title } }
            }
        }
        AND = { var:dejure_auto_manual_mode = 1
            NOR = {
                AND = { var:dejure_auto_direct_tier = 1
                    is_target_in_variable_list = { name = dejure_auto_manual_counties target = var:dejure_auto_manual_last_title } }
                AND = { var:dejure_auto_direct_tier >= 2 var:dejure_auto_direct_tier <= 4
                    is_target_in_variable_list = { name = dejure_auto_manual_titles target = var:dejure_auto_manual_last_title } }
            }
        }
    }
}
dejure_auto_direct_target_available_trigger = {
    exists = var:dejure_auto_direct_current
    exists = var:dejure_auto_direct_target
    var:dejure_auto_direct_target = {
        holder = root dejure_auto_map_title_trigger = yes
        dejure_auto_direct_target_tier_trigger = yes
    }
    var:dejure_auto_direct_current.county = {
        any_this_title_or_de_jure_above = { this = root.var:dejure_auto_direct_target }
    }
}
