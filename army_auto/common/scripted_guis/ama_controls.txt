    saved_scopes = { ama_size_object ama_size_current ama_size_max ama_size_response_ticket }
    is_valid = { ama_size_receive_trigger = yes }
    effect = { ama_size_receive_effect = yes }
}
ama_size_timeout_gui = {
    scope = character saved_scopes = { ama_size_timeout_ticket }
    is_valid = {
        ama_player_trigger = yes
        has_variable = ama_size_pending
        exists = scope:ama_size_timeout_ticket
        scope:ama_size_timeout_ticket = var:ama_size_ticket
    }
    effect = { ama_size_timeout_effect = yes }
}
ama_any_manage_gui = {
    scope = character is_shown = { OR = {
            var:ama_create_enabled = 1 var:ama_expand_enabled = 1 var:ama_title_expand_enabled = 1
            AND = { var:ama_title_create_enabled = 1 ama_title_create_available_trigger = yes }
        } }
    effect = { }
}
ama_replace_toggle_gui = { scope = character is_shown = { var:ama_replace_enabled = 1 }
    is_valid = { ama_player_trigger = yes }
    effect = { if = { limit = { ama_player_trigger = yes }
            if = { limit = { var:ama_replace_enabled = 1 } set_variable = { name = ama_replace_enabled value = 0 } }
            else = { set_variable = { name = ama_replace_enabled value = 1 } }
        } }
}

ama_repl_timeout_gui = { scope = character saved_scopes = { ama_repl_timeout_phase }
    is_valid = { has_variable = ama_repl_pending }
