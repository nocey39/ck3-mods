dau_preset_new_player_effect = {
    if = {
        limit = {
            NOT = {
                has_variable = dau_preset_player_initialized
            }
        }
        set_variable = dau_preset_player_initialized
        set_variable = dau_preset_residence_pending
        if = {
            limit = {
                has_domicile = no
            }
            set_variable = dau_preset_residence_deferred
        }
        set_variable = dau_preset_land_pending
        dau_initialize_effect = yes
        dau_land_initialize_effect = yes
    }
}
dau_preset_prepare_residence_effect = {
    if = {
        limit = {
            is_ai = no
            is_alive = yes
            NOT = {
                has_variable = dau_preset_residence_imported
            }
            NOT = {
                has_variable = dau_preset_residence_preserved
            }
            exists = scope:previous_player_character
            scope:previous_player_character = {
                is_alive = no
                has_variable = dau_preset_residence_pending
            }
        }
        set_variable = dau_preset_residence_pending
        set_variable = dau_preset_residence_deferred
    }
}
