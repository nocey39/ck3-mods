cfi_enable_culture_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds" }
    decision_group_type = cfi_management
    desc = cfi_enable_culture_decision_desc
    selection_tooltip = cfi_enable_culture_decision_tooltip
    confirm_text = cfi_enable_culture_decision_confirm
    is_shown = {
        is_ai = no
        exists = culture
        culture = { NOT = { has_cultural_tradition = tradition_cfi_integrated } }
    }
    is_valid = { is_ai = no }
    effect = {
        custom_tooltip = cfi_enable_culture_effect_tt
        hidden_effect = { cfi_enable_culture_effect = yes }
    }
    ai_check_interval = 0
    ai_will_do = { base = 0 }
}
