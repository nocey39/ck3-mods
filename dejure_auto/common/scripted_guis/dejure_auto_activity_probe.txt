dejure_auto_activity_probe_open_gui = {
    scope = character
    is_valid = { is_ai = no is_alive = yes is_landed = yes }
    effect = {
        if = { limit = { is_ai = no is_alive = yes is_landed = yes }
            set_variable = dejure_auto_activity_probe_enabled
            set_variable = { name = dejure_auto_activity_probe_records value = 0 }
            remove_variable = dejure_auto_activity_probe_county
            remove_variable = dejure_auto_activity_probe_duchy
            remove_variable = dejure_auto_activity_probe_province
            remove_variable = dejure_auto_map_probe_open
            remove_variable = dejure_auto_panel_open
        }
    }
}
dejure_auto_activity_probe_back_gui = {
    scope = character
    effect = {
        remove_variable = dejure_auto_activity_probe_enabled
        remove_variable = dejure_auto_map_probe_open
        set_variable = dejure_auto_panel_open
    }
}
dejure_auto_activity_probe_read_gui = {
    scope = character
    is_valid = {
        is_ai = no is_alive = yes is_landed = yes
        has_variable = dejure_auto_activity_probe_enabled
        exists = scope:dejure_auto_activity_probe_input
        scope:dejure_auto_activity_probe_input = {
            province_owner = root
            exists = county
            county = { tier = tier_county holder = root dejure_auto_map_county_trigger = yes }
        }
    }
    effect = {
        if = { limit = {
                is_ai = no is_alive = yes is_landed = yes
                has_variable = dejure_auto_activity_probe_enabled
                exists = scope:dejure_auto_activity_probe_input
                scope:dejure_auto_activity_probe_input = {
                    province_owner = root
                    exists = county
                    county = { tier = tier_county holder = root dejure_auto_map_county_trigger = yes }
                }
            }
            set_variable = { name = dejure_auto_activity_probe_province value = scope:dejure_auto_activity_probe_input }
            set_variable = { name = dejure_auto_activity_probe_county value = scope:dejure_auto_activity_probe_input.county }
            change_variable = { name = dejure_auto_activity_probe_records add = 1 }
            remove_variable = dejure_auto_activity_probe_duchy
            scope:dejure_auto_activity_probe_input.county = {
                every_this_title_or_de_jure_above = {
                    limit = { tier = tier_duchy }
                    save_scope_as = dejure_auto_activity_probe_found
                    root = { set_variable = { name = dejure_auto_activity_probe_duchy value = scope:dejure_auto_activity_probe_found } }
                }
            }
            clear_saved_scope = dejure_auto_activity_probe_found
        }
    }
}
