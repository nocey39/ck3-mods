# Native republic budget reenactments immediately reapply TEQ fixed salary rules.
# Suppresses republic_event.0020 for TEQ-managed assembly heads without overriding yearly_on_actions.txt.
namespace = republic_event

republic_event.0020 = {
	id_override_priority = 10
	type = character_event
	title = republic_event.0020.t
	desc = {
		desc = republic_event.0020.intro
		desc = newline.desc
		# Desc depending on if you have someone to tell you about the budget or not
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:steward
				}
				desc = republic_event.0020.steward_desc
			}
			desc = republic_event.0020.solo_desc
		}
		# Reason for needing a new budget
		first_valid = {
			triggered_desc = {
				trigger = {
					monthly_character_treasury_balance < -40
				}
				desc = republic_event.0020.deficit_desc
			}
			triggered_desc = {
				trigger = {
					character_treasury_new_budget_capacity >= { value = character_treasury_budget_capacity multiply = 1.3 }
				}
				desc = republic_event.0020.income_desc
			}
			desc = republic_event.0020.fallback_desc
		}
		triggered_desc = {
			trigger = {
				exists = scope:steward
			}
			desc = single_quotation_desc
		}
	}
	theme = stewardship
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { exists = scope:steward }
			animation = steward
		}
		triggered_animation = {
			trigger = { NOT = { exists = scope:steward } }
			animation = stressed_teacher
		}
	}
	right_portrait = {
		character = scope:steward
		animation = stressed_teacher
	}

	trigger = {
		teq_ruler_valid = no
		gdp_is_assembly_head_trigger = yes
		has_treasury = yes
		is_independent_ruler = yes
		OR = { # You are not travelling, but we don't mind if AI is
			is_ai = yes
			is_travelling = no
		}
		OR = {
			# The AI hasn't set a new budget recently
			AND = {
				is_ai = yes
				treasury_months_since_budget_enact_date >= 60
			}
			# The player hasn't set a new budget recently (time is more generous than for the AI)
			treasury_months_since_budget_enact_date >= 96
			# Left over treasury is more than 30% larger than the last budget
			character_treasury_new_budget_capacity >= { value = character_treasury_budget_capacity multiply = 1.3 }
			# You are running too big a deficit
			monthly_character_treasury_balance < -50
		}
	}

	immediate = {
		save_scope_as = treasury_ruler
		# We pick the steward, or a vassal, preferring assembly members
		if = {
			limit = { exists = cp:councillor_steward }
			cp:councillor_steward = {
				save_scope_as = steward
			}
		}
		else_if = {
			limit = {
				any_vassal = {
					count > 0
					has_same_government = liege
				}
			}
			random_vassal = {
				limit = {
					has_same_government = liege
				}
				weight = {
					base = 1
					modifier = {
						is_councillor = yes
						add = 5
					}
					modifier = {
						is_assembly_member = yes
						add = 25
					}
				}

				save_scope_as = steward
			}
		}

		if = {
			limit = {
				exists = scope:steward
			}
			random_list = {
				10 = {
					trigger = {
						prefers_high_salary_budget_trigger = { CHARACTER = scope:steward }
					}
					save_scope_as = high_salary_budget
				}
				15 = { # We skew it slightly in favor of a more balanced budget - for balancing reasons
					trigger = {
						prefers_medium_salary_budget_trigger = { CHARACTER = scope:steward }
					}
					save_scope_as = medium_salary_budget
				}
				10 = {
					trigger = {
						prefers_low_salary_budget_trigger = { CHARACTER = scope:steward }
					}
					save_scope_as = low_salary_budget
				}
			}
		}
	}

	option = { # Set a budget
		name = republic_event.0020.a

		custom_tooltip = republic_event.0020.a_tooltip
		trigger_event = republic_event.0021

		ai_chance = {
			base = 0
		}
	}

	option = { # Let the steward handle the budget for you
		trigger = {
			exists = scope:steward
		}
		name = republic_event.0020.c
		flavor = {
			triggered_desc = {
				trigger = {
					OR = {
						AND = {
							exists = scope:high_salary_budget
							has_realm_law = budget_allocation_salary_50
						}
						AND = {
							exists = scope:medium_salary_budget
							has_realm_law = budget_allocation_salary_30
						}
						AND = {
							exists = scope:low_salary_budget
							has_realm_law = budget_allocation_salary_15
						}
					}
				}
				desc = republic_event.0020.c_no_change
			}
		}

		if = { # High salary favored budget
			limit = {
				exists = scope:high_salary_budget
				NOT = { has_realm_law = budget_allocation_salary_50 }
			}
			add_realm_law = budget_allocation_salary_50
		}
		else_if = { # Medium salary favored budget
			limit = {
				exists = scope:medium_salary_budget
				NOT = { has_realm_law = budget_allocation_salary_30 }
			}
			add_realm_law = budget_allocation_salary_30
		}
		else_if = { # Low salary favored budget
			limit = {
				exists = scope:low_salary_budget
				NOT = { has_realm_law = budget_allocation_salary_15 }
			}
			add_realm_law = budget_allocation_salary_15
		}
		teq_prepare_native_budget = yes
		enact_treasury_budgets_init = yes
		teq_reassert_salary = yes

		ai_chance = {
			base = 100
		}
	}

	option = { # Keep the existing budget
		name = republic_event.0020.b

		teq_prepare_native_budget = yes
		enact_treasury_budgets_init = yes
		teq_reassert_salary = yes

		ai_chance = {
			base = 0
		}
	}
}

		else = {

		}
		teq_prepare_native_budget = yes
		enact_treasury_budgets_init = yes
		teq_reassert_salary = yes
	}
}
