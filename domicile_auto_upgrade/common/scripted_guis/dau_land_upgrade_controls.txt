dau_land_panel_gui = {
	scope = character
	is_valid = { is_ai = no }
	effect = { remove_variable = dau_land_panel_open }
}

dau_land_toggle_gui = {
	scope = character
	is_valid = {
		is_ai = no
		is_landed = yes
	}
	is_shown = { var:dau_up_enabled = 1 }
	effect = {
		if = {
			limit = { var:dau_up_enabled = 1 }
			set_variable = { name = dau_up_enabled value = 0 }
			remove_variable = dau_up_pending
		}
		else = {
			set_variable = { name = dau_up_enabled value = 1 }
			dau_upgrade_restart_effect = yes
		}
	}
}
