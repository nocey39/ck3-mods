# Character-scope getters. D/M percentages refer to their common pool, not
# salary or a native cardinal pool. Title policy is integer percentage points.
teq_control_raw_pair_weight = {
 value = { value = treasury_budget_allocation_ministries min = 0 }
 add = { value = treasury_budget_allocation_military min = 0 }
}
teq_control_raw_cardinal_weight = {
 value = 0
 if = { limit = { treasury_budget_base_rate_cardinal > 0 }
  value = treasury_budget_allocation_cardinal
 }
 min = 0
}
teq_control_seed_dev_percent = {
 value = 60
 if = { limit = { teq_control_raw_pair_weight > 0 }
  value = treasury_budget_allocation_ministries
  min = 0
  divide = teq_control_raw_pair_weight
  multiply = 100 add = 0.5 floor = yes
 }
 min = 0 max = 100
}
teq_control_development_share = {
 value = teq_control_seed_dev_percent
 if = { limit = { exists = primary_title primary_title = { has_variable = teq_control_dev_percent } }
  value = primary_title.var:teq_control_dev_percent
 }
 min = 0 max = 100 divide = 100
}
teq_control_military_share = { value = 1 subtract = teq_control_development_share min = 0 max = 1 }
teq_control_pair_weight_value = {
 value = teq_control_raw_pair_weight
 if = { limit = { exists = primary_title primary_title = { has_variable = teq_control_pair_weight } }
  value = primary_title.var:teq_control_pair_weight
 }
 min = 0
}
teq_control_cardinal_share = {
 value = 0
 if = { limit = { exists = primary_title primary_title = { has_variable = teq_control_cardinal_fraction } }
  value = primary_title.var:teq_control_cardinal_fraction
 }
 else_if = { limit = { teq_control_raw_cardinal_weight > 0 }
  value = teq_control_raw_cardinal_weight
  divide = { value = teq_control_raw_pair_weight add = teq_control_raw_cardinal_weight }
 }
 min = 0 max = 1
}
# Use raw D+M weight rather than normalized (1-C): cardinal remains untouched
# and the existing final budget scale preserves its original relative share.
teq_control_development_allocation = { value = teq_control_pair_weight_value multiply = teq_control_development_share }
teq_control_military_allocation = { value = teq_control_pair_weight_value multiply = teq_control_military_share }
teq_control_dev_offset_value = { value = 0 if = { limit = { has_variable = teq_control_dev_offset } value = var:teq_control_dev_offset } }
teq_control_mil_offset_value = { value = 0 if = { limit = { has_variable = teq_control_mil_offset } value = var:teq_control_mil_offset } }
teq_control_dev_positive_scale = { value = teq_control_dev_offset_value min = 0 }
teq_control_dev_negative_scale = { value = teq_control_dev_offset_value multiply = -1 min = 0 }
teq_control_mil_positive_scale = { value = teq_control_mil_offset_value min = 0 }
teq_control_mil_negative_scale = { value = teq_control_mil_offset_value multiply = -1 min = 0 }
