# Character scope. Each operation is guarded and safe to repeat.
cfi_enable_culture_effect = {
    culture = {
        if = {
            limit = { NOT = { has_cultural_tradition = tradition_cfi_integrated } }
            add_culture_tradition = tradition_cfi_integrated
        }
    }
}

cfi_disable_culture_effect = {
    culture = {
        if = {
            limit = { has_cultural_tradition = tradition_cfi_integrated }
            remove_culture_tradition = tradition_cfi_integrated
        }
    }
}

# 1.21 uses rites for integrated mechanics, and removes unreformed faith markers if present.
cfi_enable_religion_effect = {
    rite = {
        if = {
            limit = { NOT = { rite_has_doctrine = special_doctrine_cfi_integrated } }
            add_rite_doctrine = special_doctrine_cfi_integrated
        }
        if = {
            limit = { rite_has_doctrine = unreformed_faith_doctrine }
            remove_rite_doctrine = unreformed_faith_doctrine
        }
        if = {
            limit = { rite_has_doctrine = west_african_unreformed_faith_doctrine }
            remove_rite_doctrine = west_african_unreformed_faith_doctrine
        }
    }
    if = {
        limit = { exists = faith }
        faith = {
            if = {
                limit = { has_doctrine = unreformed_faith_doctrine }
                remove_doctrine = unreformed_faith_doctrine
            }
            if = {
                limit = { has_doctrine = west_african_unreformed_faith_doctrine }
                remove_doctrine = west_african_unreformed_faith_doctrine
            }
        }
    }
}

cfi_disable_religion_effect = {
    rite = {
        if = {
            limit = { rite_has_doctrine = special_doctrine_cfi_integrated }
            remove_rite_doctrine = special_doctrine_cfi_integrated
        }
    }
}
