dejure_auto_find_parent_lord_effect = {
    scope:dejure_auto_owner = {
        save_scope_as = dejure_auto_law_lord
        set_variable = {
            name = dejure_auto_law_found
            value = 0
        }
        set_variable = {
            name = dejure_auto_law_conflict
            value = 0
        }
    }
    if = {
        limit = {
            exists = de_jure_liege
        }
        de_jure_liege = {
            dejure_auto_scan_law_ancestors_effect = yes
        }
    }
}
dejure_auto_scan_law_ancestors_effect = {
    every_this_title_or_de_jure_above = {
        limit = {
            tier = tier_hegemony
        }
        dejure_auto_accept_law_holder_effect = yes
    }
    every_this_title_or_de_jure_above = {
        limit = {
            tier = tier_empire
        }
        dejure_auto_accept_law_holder_effect = yes
    }
    every_this_title_or_de_jure_above = {
        limit = {
            tier = tier_kingdom
        }
        dejure_auto_accept_law_holder_effect = yes
    }
    every_this_title_or_de_jure_above = {
        limit = {
            tier = tier_duchy
        }
        dejure_auto_accept_law_holder_effect = yes
    }
}
dejure_auto_accept_law_holder_effect = {
    if = {
        limit = {
            exists = holder
            dejure_auto_map_title_trigger = yes
        }
        holder = {
            if = {
                limit = {
                    this = scope:dejure_auto_owner
                }
                save_scope_as = dejure_auto_law_lord
                set_variable = {
                    name = dejure_auto_law_found
                    value = 1
                }
                set_variable = {
                    name = dejure_auto_law_conflict
                    value = 0
                }
            }
            else_if = {
                limit = {
                    is_alive = yes
                    is_landed = yes
                    is_ai = yes
                    is_vassal_or_below_of = scope:dejure_auto_owner
                    dejure_auto_force_player_branch_trigger = no
                }
                if = {
                    limit = {
                        this = scope:dejure_auto_law_lord
                    }
                    scope:dejure_auto_owner = {
                        set_variable = {
                            name = dejure_auto_law_found
                            value = 1
                        }
                    }
                }
                else_if = {
                    limit = {
                        is_vassal_or_below_of = scope:dejure_auto_law_lord
                    }
                    save_scope_as = dejure_auto_law_lord
                    scope:dejure_auto_owner = {
                        set_variable = {
                            name = dejure_auto_law_found
                            value = 1
                        }
                    }
                }
                else = {
                    scope:dejure_auto_owner = {
                        set_variable = {
                            name = dejure_auto_law_conflict
                            value = 1
                        }
                    }
                }
            }
        }
    }
}

dejure_auto_hierarchy_consider_effect = {
    save_scope_as = dejure_auto_hierarchy_person
    if = {
        limit = {
            dejure_auto_hierarchy_person_trigger = yes
        }
        primary_title = {
            dejure_auto_find_parent_lord_effect = yes
        }
        if = {
            limit = {
                dejure_auto_hierarchy_target_trigger = yes
                dejure_auto_hierarchy_subtree_safe_trigger = yes
            }
            scope:dejure_auto_owner = {
                create_title_and_vassal_change = {
                    type = granted
                    save_scope_as = dejure_auto_hierarchy_change
                    add_claim_on_loss = no
                }
            }
            change_liege = {
                liege = scope:dejure_auto_law_lord
                change = scope:dejure_auto_hierarchy_change
            }
            resolve_title_and_vassal_change = scope:dejure_auto_hierarchy_change
            clear_saved_scope = dejure_auto_hierarchy_change
            if = {
                limit = {
                    liege = scope:dejure_auto_law_lord
                }
                scope:dejure_auto_owner = {
                    change_variable = {
                        name = dejure_auto_last_moved
                        add = 1
                    }
                    change_variable = {
                        name = dejure_auto_force_used
                        add = 1
                    }
                }
                save_scope_as = dejure_auto_report_item
                scope:dejure_auto_owner = {
                    if = {
                        limit = {
                            var:dejure_auto_report_n_changed_people < 50
                            NOT = {
                                is_target_in_variable_list = {
                                    name = dejure_auto_report_changed_people
                                    target = scope:dejure_auto_report_item
                                }
                            }
                        }
                        add_to_variable_list = {
                            name = dejure_auto_report_changed_people
                            target = scope:dejure_auto_report_item
                        }
                        change_variable = {
                            name = dejure_auto_report_n_changed_people
                            add = 1
                        }
                    }
                }
                clear_saved_scope = dejure_auto_report_item
            }
            else = {
                primary_title = {
                    dejure_auto_feedback_record_unconfirmed_effect = yes
                }
            }
        }
    }
    clear_saved_scope = dejure_auto_hierarchy_person
}

        limit = {

            holder = {
                is_alive = yes
                is_ai = yes
                is_landed = yes
                is_vassal_or_below_of = scope:dejure_auto_owner
                dejure_auto_force_player_branch_trigger = no
            }
        }
        dejure_auto_find_parent_lord_effect = yes
        if = {
            limit = {
                scope:dejure_auto_owner = {
                    var:dejure_auto_law_conflict = 1
                }
            }
            scope:dejure_auto_owner = {
                change_variable = {
                    name = dejure_auto_conflicts
                    add = 1
                }
            }
            dejure_auto_record_conflict_effect = yes
        }
        if = {
            limit = {
                scope:dejure_auto_law_lord = {
                    this = scope:dejure_auto_owner
                }
                holder = {
                    highest_held_title_tier > tier_county
                }
            }
            scope:dejure_auto_owner = {
                change_variable = {
                    name = dejure_auto_unassigned
                    add = 1
                }
            }
            dejure_auto_record_missing_effect = yes
        }
        else_if = {
            limit = {
                scope:dejure_auto_law_lord = {
                    NOT = {
                        this = scope:dejure_auto_owner
                    }
                    NOT = {
                        this = scope:dejure_auto_county.holder
                    }
                    NOT = {
                        is_liege_or_above_of = scope:dejure_auto_county.holder
                    }
                }
            }
            scope:dejure_auto_owner = {
                change_variable = {
                    name = dejure_auto_left_counties
                    add = 1
                }
            }
            if = {
                limit = {
                    dejure_auto_force_county_trigger = no
                }
                dejure_auto_record_protected_effect = yes
            }
            else_if = {
                limit = {
                    dejure_auto_last_county_trigger = yes
                }
                scope:dejure_auto_owner = {
                    change_variable = {
                        name = dejure_auto_last_county_kept
                        add = 1
                    }
                }
                dejure_auto_record_last_county_effect = yes
            }
            else_if = {
                limit = {
                    scope:dejure_auto_owner = {
                        var:dejure_auto_force_used >= 50
                    }
                }
                dejure_auto_record_pending_effect = yes
            }
            else = {
                dejure_auto_record_failed_effect = yes
            }
        }
    }
    clear_saved_scope = dejure_auto_county
}
