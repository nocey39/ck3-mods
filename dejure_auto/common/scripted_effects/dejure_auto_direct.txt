    if = {
        limit = {
            NOT = {
                has_variable = dejure_auto_direct_epoch
            }

        set_variable = {
            name = dejure_auto_direct_epoch
            value = 0
        }
    }
    change_variable = {
        name = dejure_auto_direct_epoch
        add = 1
    }
    clear_variable_list = dejure_auto_direct_armed
