# Character scope throughout, except the three explicitly named title actions.
# Analytical 1-step solver using CK3 1.21 official mod-only modifier_value() —
# eliminates scriptedtests_gold_income_no_theocracy and 30-step binary-search probing,
# and calibrates ecclesiastical non-vassal treasury conversion independently.

teq_private_remove_v_modifiers = { hidden_effect = {
 remove_character_modifier = teq_private_v_positive
 remove_character_modifier = teq_private_v_negative
 remove_character_modifier = teq_private_nv_positive
 remove_character_modifier = teq_private_nv_negative
} }

teq_clear_private_split = { hidden_effect = {
 if = { limit = { OR = {
  has_variable = teq_private_split_title has_variable = teq_private_offset has_variable = teq_private_nv_offset
  has_variable = teq_private_split_month has_variable = teq_private_split_policy
  has_variable = teq_private_applied_fraction has_variable = teq_private_calibration_status
  has_character_modifier = teq_private_v_positive has_character_modifier = teq_private_v_negative
  has_character_modifier = teq_private_nv_positive has_character_modifier = teq_private_nv_negative
 } }
  teq_private_remove_v_modifiers = yes
  remove_variable = teq_private_split_title remove_variable = teq_private_split_month
  remove_variable = teq_private_split_policy remove_variable = teq_private_offset remove_variable = teq_private_nv_offset
  remove_variable = teq_private_applied_fraction remove_variable = teq_private_calibration_status
 }
} }

teq_private_calibrate = { hidden_effect = {
 set_variable = { name = teq_private_old_offset value = 0 }
 if = { limit = { has_variable = teq_private_offset has_variable = teq_private_split_title primary_title = var:teq_private_split_title }
  set_variable = { name = teq_private_old_offset value = var:teq_private_offset }
 }
 set_variable = { name = teq_private_old_nv_offset value = 0 }
 if = { limit = { has_variable = teq_private_nv_offset has_variable = teq_private_split_title primary_title = var:teq_private_split_title }
  set_variable = { name = teq_private_old_nv_offset value = var:teq_private_nv_offset }
 }

 # Remove TEQ offsets and flush once to read exact unmodified engine base & multiplier.
 teq_private_remove_v_modifiers = yes
 teq_flush_modifiers = yes

 set_variable = { name = teq_private_target_treasury value = { value = 1 subtract = teq_private_income_fraction min = 0 max = 1 } }
 if = { limit = { teq_live_vassal_treasury_denom > 0.001 }
  set_variable = {
   name = teq_private_offset
   value = {
    value = var:teq_private_target_treasury
    divide = teq_live_vassal_treasury_denom
    subtract = teq_live_vassal_treasury_base
    min = -1000 max = 1000
   }
  }
  if = { limit = { var:teq_private_offset > 0.00001 } add_character_modifier = teq_private_v_positive }
  else_if = { limit = { var:teq_private_offset < -0.00001 } add_character_modifier = teq_private_v_negative }

  # Independent calibration for 1.21 ecclesiastical non-vassal channel.
  set_variable = { name = teq_private_nv_offset value = 0 }
  if = {
   limit = {
    OR = {
     government_has_flag = government_is_ecclesiastical
     teq_live_non_vassal_treasury_base > 0
    }
    teq_live_non_vassal_treasury_denom > 0.001
   }
   set_variable = {
    name = teq_private_nv_offset
    value = {
     value = var:teq_private_target_treasury
     divide = teq_live_non_vassal_treasury_denom
     subtract = teq_live_non_vassal_treasury_base
     min = -1000 max = 1000
    }
   }
   if = { limit = { var:teq_private_nv_offset > 0.00001 } add_character_modifier = teq_private_nv_positive }
   else_if = { limit = { var:teq_private_nv_offset < -0.00001 } add_character_modifier = teq_private_nv_negative }
  }

  teq_flush_modifiers = yes
  if = { limit = { teq_private_applied_error <= 0.005 }
   set_variable = { name = teq_private_applied_fraction value = { value = 1 subtract = teq_live_vassal_treasury_share min = 0 max = 1 } }
   set_variable = { name = teq_private_calibration_status value = 1 }
  }
  else = {
   # Roll back to previous verified offset if extreme modifiers prevent exact convergence.
   teq_private_remove_v_modifiers = yes
   set_variable = { name = teq_private_offset value = var:teq_private_old_offset }
   set_variable = { name = teq_private_nv_offset value = var:teq_private_old_nv_offset }
   if = { limit = { var:teq_private_offset > 0.00001 } add_character_modifier = teq_private_v_positive }
   else_if = { limit = { var:teq_private_offset < -0.00001 } add_character_modifier = teq_private_v_negative }
   if = { limit = { var:teq_private_nv_offset > 0.00001 } add_character_modifier = teq_private_nv_positive }
   else_if = { limit = { var:teq_private_nv_offset < -0.00001 } add_character_modifier = teq_private_nv_negative }
   teq_flush_modifiers = yes
   set_variable = { name = teq_private_applied_fraction value = { value = 1 subtract = teq_live_vassal_treasury_share min = 0 max = 1 } }
   set_variable = { name = teq_private_calibration_status value = -1 }
  }
 }
 else = {
  set_variable = { name = teq_private_offset value = var:teq_private_old_offset }
  set_variable = { name = teq_private_nv_offset value = var:teq_private_old_nv_offset }
  if = { limit = { var:teq_private_offset > 0.00001 } add_character_modifier = teq_private_v_positive }
  else_if = { limit = { var:teq_private_offset < -0.00001 } add_character_modifier = teq_private_v_negative }
  if = { limit = { var:teq_private_nv_offset > 0.00001 } add_character_modifier = teq_private_nv_positive }
  else_if = { limit = { var:teq_private_nv_offset < -0.00001 } add_character_modifier = teq_private_nv_negative }
  teq_flush_modifiers = yes
  set_variable = { name = teq_private_applied_fraction value = { value = 1 subtract = teq_live_vassal_treasury_share min = 0 max = 1 } }
  set_variable = { name = teq_private_calibration_status value = -1 }
 }

 set_variable = { name = teq_private_split_title value = primary_title }
 set_variable = { name = teq_private_split_month value = teq_month_key }
 set_variable = { name = teq_private_split_policy value = teq_private_income_fraction }
 remove_variable = teq_private_old_offset
 remove_variable = teq_private_old_nv_offset
 remove_variable = teq_private_target_treasury
} }

teq_ensure_private_split = { hidden_effect = {
 if = { limit = { teq_ruler_valid = yes }
  primary_title = { if = { limit = { NOT = { has_variable = teq_private_fraction } } set_variable = { name = teq_private_fraction value = 0.05 } } }
  if = { limit = { NOT = { AND = {
   has_variable = teq_private_split_title primary_title = var:teq_private_split_title
   has_variable = teq_private_split_month var:teq_private_split_month = teq_month_key
   has_variable = teq_private_split_policy var:teq_private_split_policy = teq_private_income_fraction
   has_variable = teq_private_offset
   OR = {
    AND = { var:teq_private_offset >= -0.00001 var:teq_private_offset <= 0.00001 NOT = { has_character_modifier = teq_private_v_positive } NOT = { has_character_modifier = teq_private_v_negative } }
    AND = { var:teq_private_offset > 0.00001 has_character_modifier = teq_private_v_positive NOT = { has_character_modifier = teq_private_v_negative } }
    AND = { var:teq_private_offset < -0.00001 has_character_modifier = teq_private_v_negative NOT = { has_character_modifier = teq_private_v_positive } }
   }
  } } }
   teq_private_calibrate = yes
  }
 }
 else = { teq_clear_private_split = yes }
} }

# Title scope. Player-only, bounded, explicit directions.
teq_private_split_stepup = { hidden_effect = {
 if = { limit = { teq_fiscal_title_valid = yes holder = { is_ai = no teq_budget_ui_ready = yes teq_private_income_fraction < 1 } }
  set_variable = { name = teq_private_fraction value = { value = holder.teq_private_income_fraction add = 0.05 min = 0 max = 1 } }
  holder = { teq_ensure_private_split = yes }
  teq_apply_budget = yes
 }
} }
teq_private_split_stepdown = { hidden_effect = {
 if = { limit = { teq_fiscal_title_valid = yes holder = { is_ai = no teq_budget_ui_ready = yes teq_private_income_fraction > 0 } }
  set_variable = { name = teq_private_fraction value = { value = holder.teq_private_income_fraction subtract = 0.05 min = 0 max = 1 } }
  holder = { teq_ensure_private_split = yes }
  teq_apply_budget = yes
 }
} }
teq_private_split_restoredefault = { hidden_effect = {
 if = { limit = { teq_fiscal_title_valid = yes holder = { is_ai = no teq_budget_ui_ready = yes NOT = { teq_private_income_fraction = 0.05 } } }
  set_variable = { name = teq_private_fraction value = 0.05 }
  holder = { teq_ensure_private_split = yes }
  teq_apply_budget = yes
 }
} }
