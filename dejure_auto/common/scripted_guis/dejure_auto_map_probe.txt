dejure_auto_map_probe_open_gui = {
    scope = character
    is_valid = { is_ai = no is_alive = yes }
    effect = {
        set_variable = dejure_auto_map_probe_open
        set_variable = { name = dejure_auto_map_probe_records value = 0 }
        remove_variable = dejure_auto_map_probe_county
        remove_variable = dejure_auto_map_probe_duchy
        remove_variable = dejure_auto_panel_open
    }
}
dejure_auto_map_probe_close_gui = {
    scope = character
    effect = { remove_variable = dejure_auto_map_probe_open }
}
dejure_auto_map_probe_back_gui = {
    scope = character
    effect = { remove_variable = dejure_auto_map_probe_open set_variable = dejure_auto_panel_open }
}
dejure_auto_map_probe_read_gui = {
    scope = character
    is_valid = {
        is_ai = no is_alive = yes has_variable = dejure_auto_map_probe_open
        exists = scope:dejure_auto_map_probe_input
        scope:dejure_auto_map_probe_input = {
            tier = tier_county dejure_auto_map_county_trigger = yes
        }
    }
    effect = {
        # Defend the effect itself too; no stale previous county on invalid input.
        if = { limit = {
                is_ai = no is_alive = yes has_variable = dejure_auto_map_probe_open
                exists = scope:dejure_auto_map_probe_input
                scope:dejure_auto_map_probe_input = { tier = tier_county dejure_auto_map_county_trigger = yes }
            }
            remove_variable = dejure_auto_map_probe_duchy
            set_variable = { name = dejure_auto_map_probe_county value = scope:dejure_auto_map_probe_input }
            change_variable = { name = dejure_auto_map_probe_records add = 1 }
            scope:dejure_auto_map_probe_input = {
                every_this_title_or_de_jure_above = {
                    limit = { tier = tier_duchy }
                    save_scope_as = dejure_auto_map_probe_found
                    root = { set_variable = { name = dejure_auto_map_probe_duchy value = scope:dejure_auto_map_probe_found } }
                }
            }
            clear_saved_scope = dejure_auto_map_probe_found
        }
    }
}
