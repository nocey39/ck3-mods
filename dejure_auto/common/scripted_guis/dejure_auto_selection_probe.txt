dejure_auto_select_probe_open_gui = { scope = character
    is_valid = { is_ai = no is_alive = yes is_landed = yes NOT = { has_variable = dejure_auto_manual_open } }
    effect = { if = { limit = { is_ai = no is_alive = yes is_landed = yes NOT = { has_variable = dejure_auto_manual_open } }
            if = { limit = { NOT = { has_variable = dejure_auto_select_probe_epoch } }
                set_variable = { name = dejure_auto_select_probe_epoch value = 0 } }
            change_variable = { name = dejure_auto_select_probe_epoch add = 1 }
            set_variable = dejure_auto_select_probe_open
            remove_variable = dejure_auto_select_probe_last_1
            set_variable = { name = dejure_auto_select_probe_seen_1 value = 0 }
            remove_variable = dejure_auto_select_probe_last_2
            set_variable = { name = dejure_auto_select_probe_seen_2 value = 0 }
            remove_variable = dejure_auto_select_probe_last_3
            set_variable = { name = dejure_auto_select_probe_seen_3 value = 0 }
            remove_variable = dejure_auto_panel_open
        } }
}
