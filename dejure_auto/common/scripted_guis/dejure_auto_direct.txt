dejure_auto_direct_arm_gui = {
    scope = character
    is_valid = {
        is_ai = no
        is_alive = yes
        is_landed = yes
        has_variable = dejure_auto_manual_open
        has_variable = dejure_auto_direct_open
        exists = scope:dejure_auto_direct_reply_ticket
        var:dejure_auto_direct_epoch = scope:dejure_auto_direct_reply_ticket
        exists = scope:dejure_auto_manual_input
        scope:dejure_auto_manual_input = {
            exists = county
            county = {
                tier = tier_county
                dejure_auto_map_county_trigger = yes
                exists = holder
            }
            OR = {
                AND = {
                    root = {
                        var:dejure_auto_direct_tier = 1
                    }
                    province_owner = root
                    county = {
                        holder = root
                    }
                }
                AND = {
                    root = {
                        var:dejure_auto_direct_tier >= 2
                        var:dejure_auto_direct_tier <= 4
                    }
                    county.holder = {
                        OR = {
                            this = root
                            is_vassal_or_below_of = root
                        }
                    }
                }
            }
        }
        NOT = {
            is_target_in_variable_list = {
                name = dejure_auto_direct_armed
                target = scope:dejure_auto_manual_input
            }
        }
        NOT = {
            has_variable = dejure_auto_pick_active
        }
    }
    effect = {
        if = {
            limit = {
                is_ai = no
                is_alive = yes
                is_landed = yes
                has_variable = dejure_auto_manual_open
                has_variable = dejure_auto_direct_open
                exists = scope:dejure_auto_direct_reply_ticket
                var:dejure_auto_direct_epoch = scope:dejure_auto_direct_reply_ticket
                exists = scope:dejure_auto_manual_input
                scope:dejure_auto_manual_input = {
                    exists = county
                    county = {
                        tier = tier_county
                        dejure_auto_map_county_trigger = yes
                        exists = holder
                    }
                    OR = {
                        AND = {
                            root = {
                                var:dejure_auto_direct_tier = 1
                            }
                            province_owner = root
                            county = {
                                holder = root
                            }
                        }
                        AND = {
                            root = {
                                var:dejure_auto_direct_tier >= 2
                                var:dejure_auto_direct_tier <= 4
                            }
                            county.holder = {
                                OR = {
                                    this = root
                                    is_vassal_or_below_of = root
                                }
                            }
                        }
                    }
                }
                NOT = {
                    is_target_in_variable_list = {
                        name = dejure_auto_direct_armed
                        target = scope:dejure_auto_manual_input
                    }
                }
                NOT = {
                    has_variable = dejure_auto_pick_active
                }
            }
            add_to_variable_list = {
                name = dejure_auto_direct_armed
                target = scope:dejure_auto_manual_input
            }
            if = {
                limit = {
                    var:dejure_auto_direct_current = scope:dejure_auto_manual_input
                }
                remove_variable = dejure_auto_direct_current
                remove_variable = dejure_auto_direct_target
            }
        }
    }
}
