dejure_auto_grant_native_can_send_trigger = {
    trigger_if = {
        limit = {
            scope:puppet_or_actor = {
                highest_held_title_tier >= tier_kingdom
                has_diarchy_active_parameter = primeminister_requires_duchy
                diarch = scope:recipient
            }
        }
        custom_tooltip = {
            text = primeminister_requires_duchy.tt.blocked_grant
            any_in_list = {
                list = target_titles
                tier >= tier_duchy
            }
        }
    }
    trigger_if = {
        limit = {
            scope:actor = {
                government_has_flag = government_is_republic
            }
            scope:recipient = {
                government_has_flag = government_is_republic
            }
            scope:target = {
                is_noble_family_title = no
            }
        }
        scope:recipient = {
            trigger_if = {
                limit = {
                    is_landed = yes
                    scope:target = {
                        is_noble_family_title = no
                    }
                }
                trigger_if = {
                    limit = {
                        scope:target.tier = tier_county
                    }
                    republic_can_grant_county_to_landed_trigger = yes
                }
                trigger_else = {
                    republic_can_grant_duchy_or_higher_to_landed_trigger = yes
                }
            }
            trigger_else = {
                trigger_if = {
                    limit = {
                        scope:target.tier > tier_county
                    }
                    republic_can_grant_duchy_or_higher_to_landless_trigger = yes
                }
                trigger_else = {
                    custom_tooltip = {
                        text = republic_title_must_be_neighboring_desc
                        trigger_if = {
                            limit = {
                                any_in_list = {
                                    list = target_titles
                                    count > 1
                                }
                            }
                            any_in_list = {
                                list = target_titles
                                count = all
                                trigger_if = {
                                    limit = {
                                        tier = tier_barony
                                    }
                                    de_jure_liege = {
                                        is_in_list = target_titles
                                    }
                                }
                                trigger_else_if = {
                                    limit = {
                                        tier = tier_county
                                    }
                                    OR = {
                                        any_title_to_title_neighboring_county = {
                                            is_in_list = target_titles
                                        }
                                        any_in_de_jure_hierarchy = {
                                            tier = tier_barony
                                            is_in_list = target_titles
                                        }
                                    }
                                }
                                trigger_else = {
                                    any_in_de_jure_hierarchy = {
                                        tier = tier_county
                                        is_in_list = target_titles
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}
