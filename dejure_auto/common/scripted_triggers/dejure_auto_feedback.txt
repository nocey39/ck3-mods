dejure_auto_feedback_seat_safe_trigger = {
    save_temporary_scope_as = dejure_auto_feedback_checked_county
    holder = scope:dejure_auto_owner
    dejure_auto_map_county_trigger = yes
    is_leased_out = no
    exists = scope:dejure_auto_owner.capital_county
    scope:dejure_auto_owner.capital_county = {
        holder = scope:dejure_auto_owner
        dejure_auto_map_county_trigger = yes
    }
    NOT = {
        this = scope:dejure_auto_owner.capital_county
    }
    NOT = {
        this = scope:dejure_auto_owner.primary_title
    }
    NOT = {
        any_county_province = {
            exists = barony.holder
            barony.holder = {
                highest_held_title_tier = tier_barony
                primary_title.de_jure_liege = scope:dejure_auto_feedback_checked_county
                OR = {
                    dejure_auto_force_protected_trigger = yes
                    dejure_auto_force_player_branch_trigger = yes
                }
            }
        }
    }
}
dejure_auto_feedback_grant_source_trigger = {
    exists = scope:dejure_auto_owner.capital_county
    scope:dejure_auto_owner.capital_county = {
        holder = scope:dejure_auto_owner
        dejure_auto_map_county_trigger = yes
    }
    tier >= tier_county
    tier <= tier_empire
    holder = scope:dejure_auto_owner
    tier < scope:dejure_auto_owner.highest_held_title_tier
    tier < scope:dejure_auto_owner.primary_title.tier
    dejure_auto_map_title_trigger = yes
    is_noble_family_title = no
    is_head_of_faith = no
    is_leased_out = no
    NOT = {
        this = scope:dejure_auto_owner.primary_title
    }
    NOT = {
        this = scope:dejure_auto_owner.capital_county
    }
    trigger_if = {
        limit = {
            tier = tier_county
        }
        dejure_auto_feedback_seat_safe_trigger = yes
    }
}
