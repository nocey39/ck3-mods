# Additive hooks; old saves do not receive the new-game permit.
on_game_start = { on_actions = { dau_preset_mark_new_game } }
dau_preset_mark_new_game = {
    trigger = { NOT = { has_global_variable = game_has_started }
        NOT = { has_global_variable = dau_preset_world_initialized } }
    effect = { set_global_variable = dau_preset_new_game }
}
on_game_start_after_lobby = { on_actions = { dau_preset_after_lobby } }
dau_preset_after_lobby = {
    trigger = { has_global_variable = dau_preset_new_game
        NOT = { has_global_variable = dau_preset_world_initialized } }
    effect = {
        every_player = { limit = { is_ai = no is_alive = yes } dau_preset_new_player_effect = yes }
        set_global_variable = dau_preset_world_initialized
        remove_global_variable = dau_preset_new_game
    }
}
