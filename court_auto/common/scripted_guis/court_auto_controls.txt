court_auto_toggle_master_gui = {
    scope = character
    is_shown = { has_variable = court_auto_enabled }
    is_valid = { is_ai = no is_alive = yes }
    effect = {
        court_auto_initialize_effect = yes
        if = {
            limit = { has_variable = court_auto_enabled }
            remove_variable = court_auto_enabled
            court_auto_memory_reset_observations_effect = yes
        }
        else = { set_variable = court_auto_enabled set_variable = court_auto_skip_tasks_once court_auto_check_effect = yes }
        court_auto_restart_schedule_effect = yes
    }
}
