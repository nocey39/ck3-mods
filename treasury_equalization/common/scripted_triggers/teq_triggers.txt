# Shared native capability; neither government names nor an opt-in flag are required.
teq_uses_native_budget = { has_treasury = yes realm_uses_treasury_laws_trigger = yes }
# Adapters can override these character-scope defaults alongside their own definitions.
teq_adapter_excludes_government = { always = no }
teq_adapter_military_region = { always = no }
teq_government_valid = { teq_uses_native_budget = yes teq_adapter_excludes_government = no }
# Save-safe game rule gates: default options remain active as fallbacks on existing 0.6.x saves.
teq_scope_rule_allows = {
 OR = {
  is_ai = no
  AND = {
   NOT = { has_game_rule = teq_scope_player_only }
   trigger_if = {
    limit = { has_game_rule = teq_scope_major }
    OR = {
     highest_held_title_tier >= tier_kingdom
     sub_realm_size >= 10
    }
   }
  }
 }
}
teq_government_rule_allows = {
 trigger_if = {
  limit = { has_game_rule = teq_gov_imperial_only }
  NOT = { government_has_flag = government_is_republic }
  NOT = { government_has_flag = government_is_ecclesiastical }
 }
 trigger_else_if = {
  limit = { has_game_rule = teq_gov_no_republic }
  NOT = { government_has_flag = government_is_republic }
 }
 trigger_else_if = {
  limit = { has_game_rule = teq_gov_no_ecclesiastical }
  NOT = { government_has_flag = government_is_ecclesiastical }
 }
 trigger_else = { always = yes }
}
# Skip 0-vassal AI treasury realms (e.g. 1-county 1.21 republics) to save monthly CPU; players always qualify.
teq_ai_has_eligible_vassal = {
 OR = {
  is_ai = no
  any_vassal = {
   is_alive = yes
   is_landed = yes
   highest_held_title_tier >= tier_county
   teq_government_valid = yes
  }
 }
}
teq_ruler_valid = {
 is_alive = yes
 is_landed = yes
 teq_government_valid = yes
 # This solver reads native allocation and base-rate getters, which require independence.
 is_independent_ruler = yes
 teq_scope_rule_allows = yes
 teq_government_rule_allows = yes
 teq_ai_has_eligible_vassal = yes
}
teq_fiscal_title_valid = { exists = holder holder = { teq_ruler_valid = yes } holder.primary_title = this }
# Registration checks capabilities only, never saved flow state. Requiring an
# existing registration here would prevent a new fiscal system from starting.
teq_fiscal_chain_eligible = {
 is_alive = yes
 is_landed = yes
 # The fiscal center cannot tax or fund itself as a region.
 is_independent_ruler = no
 NOT = { this = top_liege }
 highest_held_title_tier >= tier_county
 teq_government_valid = yes
 exists = liege
 NOT = { any_liege_or_above = { teq_government_valid = no } }
 top_liege = { teq_ruler_valid = yes }
}
teq_public_tax_subject = { teq_fiscal_chain_eligible = yes }
teq_recipient_valid = {
 teq_fiscal_chain_eligible = yes
 NOT = { this = scope:teq_fiscal_title.holder }
 top_liege = scope:teq_fiscal_title.holder
}
# The hidden obligation is initialized from eligibility before this predicate
# is used. A saved marker without the matching current title/list is not ready.
teq_local_flow_registered = {
 teq_fiscal_chain_eligible = yes
 vassal_contract_has_flag = teq_public_tax_contract
 has_character_modifier = teq_public_tax
 has_variable = teq_fiscal_owner
 var:teq_fiscal_owner = top_liege.primary_title
 has_variable = teq_flow_title
 primary_title = var:teq_flow_title
 primary_title = {
  has_variable = teq_region_owner
  var:teq_region_owner = prev.var:teq_fiscal_owner
  var:teq_region_owner = {
   is_target_in_variable_list = { name = teq_regions target = prev }
  }
 }
}
teq_fiscal_center_ready = {
 teq_ruler_valid = yes
 has_character_modifier = teq_central_tax
 has_character_modifier = teq_public_collection
 has_variable = teq_policy_title
 primary_title = var:teq_policy_title
 primary_title = {
  has_variable = teq_total_budget
  has_variable = teq_policy_version
  var:teq_policy_version >= 16
 }
}
# Locks protect calculations against reentry; they must not change the tax
# income being measured by that same calculation. Zero budgets remain valid.
teq_public_flow_active = {
 teq_local_flow_registered = yes
 top_liege = { teq_fiscal_center_ready = yes }
 NOT = {
  any_liege_or_above = {
   is_independent_ruler = no
   teq_local_flow_registered = no
  }
 }
}
# Contract scope may refer to a suzerain. Only the live direct vassal relation
# within the already registered fiscal chain receives the universal obligation.
teq_public_contract_active = {
 exists = scope:subject
 exists = scope:liege
 scope:subject = {
  liege = scope:liege
  teq_public_flow_active = yes
 }
 scope:liege = {
  OR = {
   AND = {
    teq_fiscal_center_ready = yes
    primary_title = scope:subject.var:teq_fiscal_owner
   }
   AND = {
    teq_local_flow_registered = yes
    var:teq_fiscal_owner = scope:subject.var:teq_fiscal_owner
   }
  }
 }
}
teq_military_region = {
 OR = {
  teq_adapter_military_region = yes
  vassal_contract_has_flag = province_military
  vassal_contract_has_flag = celestial_grand_marshal_obligation
 }
}
teq_region_title_valid = {
 exists = holder holder.primary_title = this has_variable = teq_region_owner
 NOT = { this = var:teq_region_owner }
 exists = var:teq_region_owner.holder var:teq_region_owner = { teq_fiscal_title_valid = yes }
 holder = {
  teq_fiscal_chain_eligible = yes
  top_liege = prev.var:teq_region_owner.holder
 }
}

 teq_ruler_valid = yes
 primary_title = { has_variable = teq_total_budget has_variable = teq_policy_version var:teq_policy_version >= 9 }
}
teq_budget_owner_needs_setup = {
 teq_ruler_valid = yes
 OR = {
  NOT = { primary_title = { has_variable = teq_total_budget has_variable = teq_policy_version var:teq_policy_version >= 16 } }
  NOT = { has_variable = teq_policy_title }
  AND = { has_variable = teq_policy_title NOT = { primary_title = var:teq_policy_title } }

  NOT = { has_character_modifier = teq_public_collection }
