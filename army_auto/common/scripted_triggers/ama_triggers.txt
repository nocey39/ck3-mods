# The native `character_men_at_arms_expense_<currency>_relative` triggers log
# "Unsupported cost category type" in 1.20.1.2 once a title MAA with treasury
# upkeep exists, and the failing leg reads as false, stopping every expansion.
# Gold and treasury therefore compare the documented monthly absolute expense
# with the documented monthly income; prestige keeps the native ratio because
# the game exposes no monthly prestige income trigger.
ama_upkeep_ok_trigger = {
    OR = {
        NOT = { government_allows = use_maa_maintenance }
        var:ama_upkeep_percent = 0
        AND = {
            OR = {
                monthly_character_men_at_arms_expense_gold <= 0
                monthly_character_men_at_arms_expense_gold < ama_upkeep_budget_gold_value
            }
            OR = {
                monthly_character_men_at_arms_expense_prestige <= 0
                character_men_at_arms_expense_prestige_relative < ama_upkeep_limit_value
            }
            OR = {
                monthly_character_men_at_arms_expense_treasury <= 0
                monthly_character_men_at_arms_expense_treasury < ama_upkeep_budget_treasury_value
            }
        }
    }
}

    ama_active_trigger = yes
    has_variable = ama_size_pending
    NOT = { has_variable = ama_tx_pending }
    exists = var:ama_size_unit exists = var:ama_candidate_regiment
    exists = scope:ama_size_object exists = scope:ama_size_current
    exists = scope:ama_size_max exists = scope:ama_size_response_ticket
    scope:ama_size_response_ticket = var:ama_size_ticket
    scope:ama_size_object = var:ama_size_unit
    scope:ama_size_object = var:ama_candidate_regiment
    var:ama_candidate_mode = 2
    var:ama_candidate = var:ama_size_type
    var:ama_candidate_source = var:ama_size_source
    var:ama_candidate_category = var:ama_size_category

        var:ama_size_source = 0
        AND = { exists = var:ama_size_title exists = var:ama_candidate_title
            var:ama_size_title = var:ama_candidate_title }
    }
}

    exists = scope:ama_size_max exists = scope:ama_size_response_ticket

    scope:ama_size_object = var:ama_candidate_regiment
    scope:ama_size_object = var:ama_size_unit

    scope:ama_size_max > 0
    scope:ama_size_object.ama_size_next_level_value <= scope:ama_size_max
}
