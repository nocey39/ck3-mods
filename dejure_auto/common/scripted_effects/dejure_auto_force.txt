                    is_ai = yes

                        liege = scope:dejure_auto_force_recipient

                }

                }
            }
        }
    }
    resolve_title_and_vassal_change = scope:dejure_auto_force_change
    clear_saved_scope = dejure_auto_force_change
    if = {
        limit = {
            holder = scope:dejure_auto_force_recipient

                name = dejure_auto_last_taken

            }

                    var:dejure_auto_report_n_changed_titles < 50

                    target = scope:dejure_auto_report_item

                change_variable = {
                    name = dejure_auto_report_n_changed_titles
                    add = 1
                }
            }
        }
        clear_saved_scope = dejure_auto_report_item
    }
