dau_panel_gui = {
    scope = character
    is_shown = { has_variable = dau_panel_open }
    effect = { remove_variable = dau_panel_open }
}
dau_toggle_gui = {
    scope = character
    is_shown = { domicile ?= { has_variable = dau_enabled } }
    is_valid = { is_ai = no has_domicile = yes dau_supported_domicile_trigger = yes }
    effect = {
        dau_initialize_effect = yes
        if = {
            limit = { domicile = { has_variable = dau_enabled } }
            domicile = { remove_variable = dau_enabled }
        }
        else = {
            domicile = { set_variable = dau_enabled set_variable = { name = dau_elapsed value = 0 } }
            dau_check_effect = yes
        }
        dau_save_settings_effect = yes
        dau_restart_monthly_effect = yes
    }
}
