namespace = adl_choices

adl_choices.0001 = {
	type = character_event
	theme = dynasty
	title = ce1_heroic_legacy_3_name
	desc = adl_choose_trait_desc
	option = {
		name = trait_herald
		trigger = {
			is_ai = no
			exists = scope:adl_choice_dynasty
			dynasty = scope:adl_choice_dynasty
			adl_can_choose_ce1_heroic_legacy_3_trigger = yes
		}
		hidden_effect = {
			remove_character_flag = adl_choice_open
			if = {
				limit = {
					is_ai = no
					exists = scope:adl_choice_dynasty
					dynasty = scope:adl_choice_dynasty
					adl_can_choose_ce1_heroic_legacy_3_trigger = yes
				}
				dynasty = {
					all_dynasty_legacy_perk_effect = { PERK = ce1_heroic_legacy_3 }
					if = {
						limit = { has_dynasty_perk = ce1_heroic_legacy_3 }
						set_variable = { name = ce1_heroic_legacy_3_herald value = yes }
					}
				}
				if = { limit = { dynasty = { adl_selected_ce1_heroic_legacy_3_trigger = yes } } all_dynasty_legacies_effect = yes }
			}
		}
	}
	option = {
		name = adl_choose_later
		hidden_effect = { remove_character_flag = adl_choice_open }
	}
}
