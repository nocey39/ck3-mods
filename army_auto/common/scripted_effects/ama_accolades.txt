
ama_acc_clear_request_effect = {
    remove_variable = ama_acc_target
    remove_variable = ama_acc_target_type
    remove_variable = ama_acc_request
    clear_variable_list = ama_acc_requests_0
    clear_variable_list = ama_acc_requests_1
}
ama_acc_check_effect = {
    ama_acc_clear_request_effect = yes
    if = {
        limit = { has_variable = ama_acc_enabled ama_acc_available_trigger = yes }
        save_scope_as = owner
        save_scope_as = value_target
        ama_acc_row_effect = { SLOT = 1 }
        ama_acc_row_effect = { SLOT = 2 }
        ama_acc_row_effect = { SLOT = 3 }
        ama_acc_row_effect = { SLOT = 4 }
        ama_acc_row_effect = { SLOT = 5 }
        if = {
            limit = { has_variable = ama_acc_request }
            # Alternate the host lists to give each request a fresh GUI animation.
            if = {
                limit = { var:ama_acc_request_phase ?= 1 }
                set_variable = { name = ama_acc_request_phase value = 0 }
                add_to_variable_list = { name = ama_acc_requests_0 target = var:ama_acc_target }
            }
            else = {
                set_variable = { name = ama_acc_request_phase value = 1 }
                add_to_variable_list = { name = ama_acc_requests_1 target = var:ama_acc_target }
            }
        }
    }
}
ama_acc_row_effect = {
    if = {
        limit = {
            has_variable = ama_acc_first_$SLOT$
            has_variable = ama_acc_second_$SLOT$
        }
        if = {
            limit = {
                any_accolade = {
                    ama_acc_initial_type_trigger = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ }
                }
            }
            every_accolade = {
                limit = { ama_acc_initial_type_trigger = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ } }
                ama_acc_plan_upgrade_effect = { SLOT = $SLOT$ }
            }
            set_variable = { name = ama_acc_state_$SLOT$ value = 1 }
        }
        else_if = {
            limit = {
                OR = {
                    any_accolade = { ama_acc_has_type_trigger = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ } }
                    any_accolade = { ama_acc_has_type_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ } }
                }
            }
            set_variable = { name = ama_acc_state_$SLOT$ value = 2 }
        }
        else_if = {
            limit = { num_accolades >= max_accolades }
            set_variable = { name = ama_acc_state_$SLOT$ value = 3 }
        }
        else_if = {
            limit = {
                NOT = {
                    ama_acc_culture_trigger = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ }
                    ama_acc_culture_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ }
                }
            }
            set_variable = { name = ama_acc_state_$SLOT$ value = 4 }
        }
        else_if = {
            limit = { prestige < accolade_creation_prestige_cost }
            set_variable = { name = ama_acc_state_$SLOT$ value = 5 }
        }
        else_if = {
            limit = { NOT = { has_variable = ama_acc_retry_$SLOT$ } }
            set_variable = { name = ama_acc_state_$SLOT$ value = 6 }
            ama_acc_generate_effect = { SLOT = $SLOT$ }
            if = {
                limit = {
                    any_accolade = { ama_acc_initial_type_trigger = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ } }
                }
                set_variable = { name = ama_acc_state_$SLOT$ value = 1 }
            }
            else = {
                set_variable = { name = ama_acc_retry_$SLOT$ value = yes days = 365 }
            }
        }
        else = {
            # Preserve a specific failure while refreshing obsolete generic states.
            if = {
                limit = { NOT = { OR = {
                    var:ama_acc_state_$SLOT$ ?= 8
                    var:ama_acc_state_$SLOT$ ?= 9
                    var:ama_acc_state_$SLOT$ ?= 10
                    var:ama_acc_state_$SLOT$ ?= 11
                } } }
                set_variable = { name = ama_acc_state_$SLOT$ value = 6 }
            }
        }
    }
}
ama_acc_generate_effect = {
    if = {
        limit = {
            OR = {
                scope:owner.var:ama_acc_first_$SLOT$ = flag:contender_attribute
                scope:owner.var:ama_acc_second_$SLOT$ = flag:contender_attribute
            }
        }
        ama_acc_spawn_effect = { PROWESS = 24 }
    }
    else = { ama_acc_spawn_effect = { PROWESS = 15 } }
    scope:ama_acc_candidate = {
        ama_acc_train_effect = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ }
        ama_acc_train_effect = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ }
    }
    if = {
        limit = { scope:ama_acc_candidate = { NOT = { can_be_knight_trigger = { ARMY_OWNER = scope:owner } } } }
        set_variable = { name = ama_acc_state_$SLOT$ value = 8 }
    }
    else_if = {
        limit = { scope:ama_acc_candidate = { NOT = { ama_acc_fit_trigger = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ } } } }
        set_variable = { name = ama_acc_state_$SLOT$ value = 9 }
    }
    else_if = {
        limit = { scope:ama_acc_candidate = { NOT = { ama_acc_fit_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ } } } }
        set_variable = { name = ama_acc_state_$SLOT$ value = 10 }
    }
    else = {
        scope:ama_acc_candidate = { set_knight_status = force }
        ama_acc_create_effect = { SLOT = $SLOT$ }
        if = {
            limit = { scope:ama_acc_candidate = { NOT = { exists = accolade } } }
            set_variable = { name = ama_acc_state_$SLOT$ value = 11 }
        }
    }
    if = {
        limit = { scope:ama_acc_candidate = { NOT = { exists = accolade } } }
        scope:ama_acc_candidate = { set_knight_status = default move_to_pool = yes }
    }
}

ama_acc_plan_upgrade_effect = {
    if = {
        limit = {
            exists = acclaimed_knight
            accolade_can_spend_attribute = yes
            accolade_owner = { NOT = { has_variable = ama_acc_request } }
        }
        save_temporary_scope_as = ama_acc_current
        if = {
            limit = {
                NOT = { ama_acc_has_type_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ } }
                accolade_can_have_more_attributes = yes
                acclaimed_knight = { ama_acc_fit_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ } }
                accolade_owner = {
                    NOT = { any_accolade = { ama_acc_has_type_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ } } }
                }
            }
            accolade_owner = {
                set_variable = { name = ama_acc_target value = scope:ama_acc_current }
                set_variable = { name = ama_acc_target_type value = scope:owner.var:ama_acc_second_$SLOT$ }
                set_variable = { name = ama_acc_request value = 1 }
            }
        }
        else_if = {
            limit = { ama_acc_has_type_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ } }
            if = {
                limit = { ama_acc_type_below_max_trigger = { TYPE = scope:owner.var:ama_acc_first_$SLOT$ } }
                accolade_owner = {
                    set_variable = { name = ama_acc_target value = scope:ama_acc_current }
                    set_variable = { name = ama_acc_target_type value = scope:owner.var:ama_acc_first_$SLOT$ }
                    set_variable = { name = ama_acc_request value = 2 }
                }
            }
            else_if = {
                limit = { ama_acc_type_below_max_trigger = { TYPE = scope:owner.var:ama_acc_second_$SLOT$ } }
                accolade_owner = {
                    set_variable = { name = ama_acc_target value = scope:ama_acc_current }
                    set_variable = { name = ama_acc_target_type value = scope:owner.var:ama_acc_second_$SLOT$ }
                    set_variable = { name = ama_acc_request value = 2 }
                }
            }
        }
    }
}
