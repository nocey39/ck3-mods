fa_invalidate_guardian_schedule_effect = {
    remove_variable = fa_guardian_pending
    remove_variable = fa_guardian_check_pending
    if = {
        limit = { NOT = { has_variable = fa_guardian_generation } }
        set_variable = { name = fa_guardian_generation value = 0 }
    }
    change_variable = { name = fa_guardian_generation add = 1 }
}

fa_schedule_guardian_check_effect = {
    if = {
        limit = { fa_guardian_active_trigger = yes NOT = { has_variable = fa_guardian_check_pending } }
        fa_invalidate_guardian_schedule_effect = yes
        set_variable = { name = fa_guardian_check_pending value = yes days = 45 }
        save_scope_value_as = { name = fa_guardian_ticket value = var:fa_guardian_generation }
        trigger_event = { on_action = fa_scheduled_guardian_check months = 1 }
    }
}
