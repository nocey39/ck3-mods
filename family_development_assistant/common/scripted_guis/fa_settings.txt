}

fa_check_now_gui = {
    scope = character
    is_valid = {
        OR = {
            fa_guardian_active_trigger = yes
            AND = { fa_perk_active_trigger = yes is_adult = yes }
            fa_marriage_active_trigger = yes
        }
    }
    effect = { fa_check_enabled_modules_effect = yes }
}

fa_guardians_enabled_gui = {
    scope = character
    is_shown = { has_variable = fa_guardians_enabled }
    is_valid = { fa_can_configure_trigger = yes }
    effect = {
        if = {
            limit = { has_variable = fa_guardians_enabled }
            remove_variable = fa_guardians_enabled
            fa_invalidate_guardian_schedule_effect = yes
        }
        else = {
            set_variable = fa_guardians_enabled
            if = {
                limit = { fa_guardian_active_trigger = yes }
                fa_check_guardians_effect = yes
                fa_schedule_guardian_check_effect = yes
            }
        }
    }
}
