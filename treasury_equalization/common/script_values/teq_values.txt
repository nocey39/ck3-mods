# Net-after-expenses treasury split (Plan A):
# Private gold split p% is applied pre-treasury by the engine as before.
# From the resulting treasury income, deduct all central non-budget expenses
# (unraised/raised Men-at-Arms, court positions, mercenaries, etc.) to get:
# NetSpendable = max(character_treasury_new_budget_capacity, 0).
# Central treasury reserve = NetSpendable * r (default 5%),
# Distributed vassal budget = NetSpendable * (1 - r).
teq_retained_income_fraction = {
 value = 0.05
 if = { limit = { primary_title = { has_variable = teq_retained_fraction } }
  value = primary_title.var:teq_retained_fraction
 }
 min = 0 max = 1
}
teq_treasury_share_of_gross = { value = 1 subtract = teq_private_income_fraction min = 0 max = 1 }
teq_budget_share_of_treasury = {
 value = 1
 subtract = teq_retained_income_fraction
 min = 0
 max = 1
}
teq_net_spendable_treasury = {
 value = character_treasury_new_budget_capacity
 if = {
  limit = { exists = primary_title primary_title = { has_variable = teq_available_flow } }
  value = primary_title.var:teq_available_flow
 }
 min = 0
}
teq_budget_ceiling = { value = teq_net_spendable_treasury multiply = teq_budget_share_of_treasury min = 0 }
