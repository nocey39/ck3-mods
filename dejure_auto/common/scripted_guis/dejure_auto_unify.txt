dejure_auto_unify_faith_gui = {
    scope = character is_shown = { var:dejure_auto_unify_force_faith = 1 }
    is_valid = { is_ai = no is_alive = yes is_landed = yes }
    effect = { dejure_auto_initialize_effect = yes
        if = { limit = { var:dejure_auto_unify_force_faith = 1 } set_variable = { name = dejure_auto_unify_force_faith value = 0 } }
        else = { set_variable = { name = dejure_auto_unify_force_faith value = 1 } }
    }
}
