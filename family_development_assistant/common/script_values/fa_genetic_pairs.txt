# Pair estimates, derived from vanilla's inheritance chances. These are ranking values.
fa_genetic_intellect_positive = {
    value = 0.008
    if = { limit = { has_trait = intellect_good scope:fa_marriage_subject = { has_trait = intellect_good } } value = 0.8 }
    else_if = { limit = { OR = { AND = { has_trait = intellect_good scope:fa_marriage_subject = { has_inactive_trait = intellect_good } } AND = { has_inactive_trait = intellect_good scope:fa_marriage_subject = { has_trait = intellect_good } } } } value = 0.5 }
    else_if = { limit = { OR = { has_trait = intellect_good scope:fa_marriage_subject = { has_trait = intellect_good } } } value = 0.25 }
    else_if = { limit = { has_inactive_trait = intellect_good scope:fa_marriage_subject = { has_inactive_trait = intellect_good } } value = 0.1 }
    else_if = { limit = { OR = { has_inactive_trait = intellect_good scope:fa_marriage_subject = { has_inactive_trait = intellect_good } } } value = 0.02 }
}

fa_genetic_pair_score = {
    value = 0
    if = {
        limit = { scope:fa_marriage_subject.dynasty ?= { has_dynasty_modifier = strong_blood } }
        add = {
        value = fa_genetic_intellect_top
        multiply = fa_genetic_physique_top
        multiply = fa_genetic_beauty_top
        multiply = fa_genetic_pure_blooded
        multiply = fa_genetic_fecund
            multiply = 250
        }
        add = {
        value = fa_genetic_intellect_top
        multiply = fa_genetic_physique_top
        multiply = fa_genetic_beauty_top
            multiply = 60
        }
    }
    else = {
        add = {
        value = fa_genetic_intellect_positive
        multiply = fa_genetic_physique_positive
        multiply = fa_genetic_beauty_positive
            subtract = {
        value = fa_genetic_intellect_below_top
        multiply = fa_genetic_physique_below_top
        multiply = fa_genetic_beauty_below_top
            }
            multiply = 600
        }
    }
    # Scores still distinguish pairs when reaching all of the current targets is unlikely.
    add = { value = fa_genetic_intellect_positive multiply = 2 }
    add = { value = fa_genetic_physique_positive multiply = 2 }
    add = { value = fa_genetic_beauty_positive multiply = 2 }
    add = { value = fa_genetic_intellect_top multiply = 2 }
    add = { value = fa_genetic_physique_top multiply = 2 }
    add = { value = fa_genetic_beauty_top multiply = 2 }
    add = { value = fa_genetic_pure_blooded multiply = 3 }
    add = { value = fa_genetic_fecund multiply = 2 }
    multiply = {
        value = fa_genetic_fertility_estimate
        add = scope:fa_marriage_subject.fa_genetic_fertility_estimate
        divide = 2
    }
}
