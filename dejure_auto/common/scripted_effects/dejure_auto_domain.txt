dejure_auto_domain_snapshot_effect = {
    every_held_title = { limit = { dejure_auto_map_county_trigger = yes holder = scope:dejure_auto_owner }
        add_to_temporary_list = dejure_auto_domain_counties }
}
dejure_auto_domain_normal_effect = {
    # Normal transfers have already consumed part of this same 50-change budget.
    set_variable = { name = dejure_auto_force_used value = var:dejure_auto_last_moved }
    dejure_auto_domain_snapshot_effect = yes
    every_in_list = { list = dejure_auto_domain_counties
        limit = { scope:dejure_auto_owner = { var:dejure_auto_force_used < 50 } }
        dejure_auto_domain_consider_effect = yes }
    every_in_list = { list = dejure_auto_domain_counties dejure_auto_domain_report_effect = yes }
    clear_saved_scope = dejure_auto_law_lord
    remove_variable = dejure_auto_force_used
    remove_variable = dejure_auto_law_found
    remove_variable = dejure_auto_law_conflict
}
dejure_auto_domain_consider_effect = {
    save_scope_as = dejure_auto_county
    if = { limit = { dejure_auto_domain_safe_trigger = yes }
        dejure_auto_find_parent_lord_effect = yes
        if = { limit = { dejure_auto_domain_recipient_trigger = yes
                dejure_auto_domain_safe_trigger = yes
                scope:dejure_auto_owner = { var:dejure_auto_force_used < 50 } }
            scope:dejure_auto_law_lord = { save_scope_as = dejure_auto_force_recipient }
            dejure_auto_force_transfer_effect = yes
            if = { limit = { holder = scope:dejure_auto_force_recipient }
                scope:dejure_auto_owner = { change_variable = { name = dejure_auto_last_player_counties add = 1 } }
            }
        }
    }
    clear_saved_scope = dejure_auto_force_recipient
    clear_saved_scope = dejure_auto_county
}
dejure_auto_domain_report_effect = {
    save_scope_as = dejure_auto_county
    if = { limit = { dejure_auto_domain_source_trigger = yes }
        if = { limit = { dejure_auto_domain_safe_trigger = no } dejure_auto_record_protected_effect = yes }
        else = {
            dejure_auto_find_parent_lord_effect = yes
            if = { limit = { dejure_auto_domain_recipient_trigger = no }
                scope:dejure_auto_owner = { change_variable = { name = dejure_auto_unassigned add = 1 } }
                dejure_auto_record_missing_effect = yes
            }
            else = {
                scope:dejure_auto_owner = { change_variable = { name = dejure_auto_left_counties add = 1 } }
                if = { limit = { scope:dejure_auto_owner = { var:dejure_auto_force_used >= 50 } } dejure_auto_record_pending_effect = yes }
                else = { dejure_auto_record_failed_effect = yes }
            }
        }
    }
    clear_saved_scope = dejure_auto_county
    clear_saved_scope = dejure_auto_law_lord
}
