court_auto_review_initialize_effect = {
    if = { limit = { NOT = { has_variable = court_auto_review_interval } } set_variable = { name = court_auto_review_interval value = 1 } }
    if = { limit = { NOT = { has_variable = court_auto_review_elapsed } } set_variable = { name = court_auto_review_elapsed value = 0 } }
    if = { limit = { NOT = { has_variable = court_auto_council_replace } } set_variable = { name = court_auto_council_replace value = 0 } }
    if = { limit = { NOT = { has_variable = court_auto_court_replace } } set_variable = { name = court_auto_court_replace value = 0 } }
}

    set_variable = { name = court_auto_review_interval value = 1 }
    set_variable = { name = court_auto_review_elapsed value = 0 }
    set_variable = { name = court_auto_council_replace value = 0 }

court_auto_replace_court_type_effect = {
    remove_variable = court_auto_review_stop
    set_variable = { name = court_auto_review_attempts value = 0 }
    while = {
        limit = {
            employs_court_position = $POSITION$
            NOT = { has_variable = court_auto_review_stop }
            var:court_auto_review_attempts < 100
            court_auto_review_$POSITION$_affordable_trigger = yes
        }
        change_variable = { name = court_auto_review_attempts add = 1 }
        remove_variable = court_auto_review_candidate
        remove_variable = court_auto_review_holder
        ordered_court_position_holder = {
            type = $POSITION$
            order_by = { value = aptitude_score:$POSITION$ multiply = -1 }
            max = 1 check_range_bounds = no
            scope:court_auto_owner = {
                set_variable = { name = court_auto_review_holder value = prev }
                set_variable = { name = court_auto_review_score value = prev.aptitude_score:$POSITION$ }
            }
        }
        if = { limit = { has_variable = court_auto_review_holder }
            ordered_court_position_candidate = {
                court_position_type = $POSITION$
                order_by = aptitude_score:$POSITION$
                max = 1 check_range_bounds = no
                limit = {
                    NOT = { has_court_position = $POSITION$ }
                    would_be_valid_for_court_position = { court_position = $POSITION$ employer = scope:court_auto_owner }
                    aptitude_score:$POSITION$ > scope:court_auto_owner.var:court_auto_review_score
                }
                scope:court_auto_owner = { set_variable = { name = court_auto_review_candidate value = prev } }
            }
        }
        if = { limit = {
                has_variable = court_auto_review_candidate
                any_court_position_holder = { type = $POSITION$ this = scope:court_auto_owner.var:court_auto_review_holder }
                var:court_auto_review_candidate = {
                    NOT = { has_court_position = $POSITION$ }
                    would_be_valid_for_court_position = { court_position = $POSITION$ employer = scope:court_auto_owner }
                    aptitude_score:$POSITION$ > scope:court_auto_owner.var:court_auto_review_score
                }
                court_auto_review_$POSITION$_affordable_trigger = yes
            }
            replace_court_position = { recipient = var:court_auto_review_candidate holder = var:court_auto_review_holder court_position = $POSITION$ }
            if = { limit = { any_court_position_holder = { type = $POSITION$ this = scope:court_auto_owner.var:court_auto_review_candidate } }
                change_variable = { name = court_auto_last_court add = 1 }
            }
            else = { set_variable = court_auto_review_stop }
        }
        else = { set_variable = court_auto_review_stop }
    }
    remove_variable = court_auto_review_candidate
    remove_variable = court_auto_review_holder
    remove_variable = court_auto_review_score
    remove_variable = court_auto_review_stop
    remove_variable = court_auto_review_attempts
}
